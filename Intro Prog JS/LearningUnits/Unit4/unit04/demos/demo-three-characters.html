Three Character Program

Problem Statement

Design a program that will prompt for three characters, accept those characters as input, sort them into ascending sequence and output them to the screen. The program is to continue to read characters until XXX, XXX, XXX is entered.

Planning List

Input Variables

  • character1
  • character2
  • character3

Output Variables

  • character1
  • character2
  • character3

Process Checklist

  1. prompt for characters
  2. accept three characters
  3. sort three characters
  4. output three characters

Version 1

The initial program is all in one function.

function processThreeCharactersVersion1() {
	"use strict";

	let character1AllInOne;
	let character2AllInOne;
	let character3AllInOne;
	let swapAllInOne;

    character1AllInOne = prompt("Enter one character (1/3) "
			+ "or 'XXX' to quit.");
    character2AllInOne = prompt("Enter one character (2/3) "
			+ "or 'XXX' to quit.");
    character3AllInOne = prompt("Enter one character (3/3) "
			+ "or 'XXX' to quit.");

    while (character1AllInOne !== "XXX"
			&& character2AllInOne !== "XXX"
			&& character3AllInOne !== "XXX") {

		if (character1AllInOne > character2AllInOne) {
			swapAllInOne = character1AllInOne;
			character1AllInOne = character2AllInOne;
			character2AllInOne = swapAllInOne;
		}

		if (character2AllInOne > character3AllInOne) {
			swapAllInOne = character2AllInOne;
			character2AllInOne = character3AllInOne;
			character3AllInOne = swapAllInOne;
		}

		if (character1AllInOne > character2AllInOne) {
			swapAllInOne = character1AllInOne;
			character1AllInOne = character2AllInOne;
			character2AllInOne = swapAllInOne;
		}

		document.write("character1: " + character1AllInOne + "
");
		document.write("character2: " + character2AllInOne + "
");
		document.write("character3: " + character3AllInOne + "
");

		character1AllInOne = prompt("Enter one character (1/3) "
				+ "or 'XXX' to quit.");
		character2AllInOne = prompt("Enter one character (2/3) "
				+ "or 'XXX' to quit.");
		character3AllInOne = prompt("Enter one character (3/3) "
				+ "or 'XXX' to quit.");
    }
}

Version 2

To begin modularizing this program, we can start by reducing the size of the loop. Keeping loop code small is always a good idea. When it’s small we can better see errors and follow the logic.

We’re going to just take the if statements and put them into another function and replace them with a call to the new function.

Since each function needs access to the variable, we also need to make the variables global. We will learn more about that in the next lecture, but what this means is that we will be declaring our variables at the top of the file, before any functions are declared. This allows each function in the file to access the data inside those variables.

// Global Variables
let character1;
let character2;
let character3;
let swap;
	
/*
	The purpose of this function is to
	sort the three characters that were entered by the viewer.
*/
function sortThreeCharacters() {
	"use strict";

	if (character1 > character2) {
		swap = character1;
		character1 = character2;
		character2 = swap;
	}

	if (character2 > character3) {
		swap = character2;
		character2 = character3;
		character3 = swap;
	}

	if (character1 > character2) {
		swap = character1;
		character1 = character2;
		character2 = swap;
	}
}
	
/*
	The purpose of this function is to control the
	Three Characters program.
*/
function processThreeCharactersVersion2() {
	"use strict";

	character1 = prompt("Enter one character (1/3) "
			+ "or 'XXX' to quit.");
	character2 = prompt("Enter one character (2/3) "
			+ "or 'XXX' to quit.");

}

Version 3

Do you see any duplicate code in Version 2? We need to eliminate duplicate code to keep our program DRY. DRY stands for Don't Repeate Yourself.

Also, it’s a really good idea to have our controller function to not have much lower-level code. So, let’s take out everything and replace it with function calls.

// Global Variables
let character1;
let character2;
let character3;
let swap;

/*
	The purpose of this function is to
	sort the three characters that were entered by the viewer.
*/
function sortThreeCharacters() {
	"use strict";

	if (character1 > character2) {
		swap = character1;
		character1 = character2;
		character2 = swap;
	}

	if (character2 > character3) {
		swap = character2;
		character2 = character3;
		character3 = swap;
	}

	if (character1 > character2) {
		swap = character1;
		character1 = character2;
		character2 = swap;
	}
}

/*
	The purpose of this function is to
	receive the input of three characters.
*/
function readThreeCharacters() {
	"use strict";

	character1 = prompt("Enter one character (1 of 3) "
			+ "or 'XXX' to quit.");
	character2 = prompt("Enter one character (2 of 3) "
			+ "or 'XXX' to quit.");
	character3 = prompt("Enter one character (3 of 3) "
			+ "or 'XXX' to quit.");
}

/*
	The purpose of this function is to
	display the three characters on the Web page.
*/
function printThreeCharacters() {
	"use strict";

	document.write("character1: " + character1 + "<br />");
	document.write("character2: " + character2 + "<br />");
	document.write("character3: " + character3 + "<br />");
}

/*
	The purpose of this main function is to
	coordinate the overall processing required to
	repeated read, sort, and display three characters
	input by the viewer.
*/
function processThreeCharactersVersion3() {
	"use strict";

	readThreeCharacters();

	while (character1 !== 'XXX'
			&& character2 !== 'XXX'
			&& character1 !== 'XXX') {
		sortThreeCharacters();
		printThreeCharacters();
		readThreeCharacters();
	}
}
